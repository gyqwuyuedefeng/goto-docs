def _font_usage(scene, prepared):
    index, used = {}, {}
    for _, meta in prepared:
        key = (meta['family'], meta['slot'])
        digest = meta['sha256']
        if key in index and index[key] != digest:
            raise ValueError('同名字体资源存在歧义')
        index[key] = digest
        used.setdefault(digest, set())
    for item in scene['objects']:
        if item['kind'] != 'text':
            continue
        for run in item['runs']:
            key = (run['font_family'], 'bold' if run['bold'] else 'regular')
            if key not in index:
                raise ValueError('文本引用缺少已验证字体')
            used[index[key]].update(map(ord, run['text']))
    return used

def _build_font_subset(raw, codepoints):
    with TTFont(BytesIO(raw), recalcTimestamp=False) as font:
        options = font_subset.Options()
        options.layout_features = ['*']
        options.name_IDs = ['*']
        options.name_languages = ['*']
        options.name_legacy = True
        options.glyph_names = True
        options.legacy_cmap = True
        options.symbol_cmap = True

        options.notdef_outline = True

        options.hinting = True
        options.recalc_bounds = False
        options.recalc_timestamp = False
        subsetter = font_subset.Subsetter(options=options)
        subsetter.populate(unicodes=sorted(codepoints))
        subsetter.subset(font)
        output = BytesIO()
        font.save(output)
        return output.getvalue()


def _validate_font_subset(source_raw, subset_raw, codepoints):
    with TTFont(BytesIO(source_raw)) as source, TTFont(BytesIO(subset_raw)) as reduced:
        for table, fields in (
            ('head', ('unitsPerEm',)),
            ('hhea', ('ascent', 'descent', 'lineGap')),
            ('OS/2', ('fsType', 'fsSelection', 'usWeightClass', 'usWidthClass',
                      'sTypoAscender', 'sTypoDescender', 'sTypoLineGap',
                      'usWinAscent', 'usWinDescent')),
        ):
            for field in fields:
                if getattr(source[table], field) != getattr(reduced[table], field):
                    raise ValueError('子集字体许可、样式或度量改变')
        def names(font):
            return {(r.nameID, r.platformID, r.platEncID, r.langID): r.toUnicode()
                    for r in font['name'].names}
        if names(source) != names(reduced):
            raise ValueError('子集字体名称改变')
        original_map, reduced_map = source.getBestCmap(), reduced.getBestCmap()
        if not codepoints <= set(original_map) or not codepoints <= set(reduced_map):
            raise ValueError('子集字体缺少使用字符')
        for codepoint in codepoints:
            if original_map[codepoint] != reduced_map[codepoint]:
                raise ValueError('子集字符映射改变')
        source_glyphs, reduced_glyphs = source.getGlyphSet(), reduced.getGlyphSet()
        for name in reduced.getGlyphOrder():
            if name not in source_glyphs or name not in source['hmtx'].metrics:
                raise ValueError('子集关联字形身份改变')
            if source['hmtx'][name] != reduced['hmtx'][name]:
                raise ValueError('子集字形字宽改变')
            left = DecomposingRecordingPen(source_glyphs, skipMissingComponents=False)
            right = DecomposingRecordingPen(reduced_glyphs, skipMissingComponents=False)
            source_glyphs[name].draw(left)
            reduced_glyphs[name].draw(right)
            if left.value != right.value:
                raise ValueError('子集字形轮廓改变')
        if 'glyf' not in reduced or 'fvar' in reduced:
            raise ValueError('子集不再是静态TrueType')


def _deflated_size(data):
    encoder = zlib.compressobj(wbits=-15)
    return len(encoder.compress(data) + encoder.flush())

        raw_size = struct.unpack_from('<I', full_eot, 4)[0]
        if not 0 < raw_size <= len(full_eot):
            raise ValueError('完整字体载荷长度无效')
        raw = full_eot[-raw_size:]
        if hashlib.sha256(raw).hexdigest() != original['sha256']:
            raise ValueError('完整字体载荷与已验证身份不符')
        full_meta = dict(original, source_font_bytes=len(raw),
                         embedded_font_sha256=original['sha256'],
                         source_eot_bytes=len(full_eot),
                         source_deflated_bytes=_deflated_size(full_eot),
                         embedded_deflated_bytes=_deflated_size(full_eot),
                         embedding='full', is_subset=False, subset_reason='not_attempted')
        def keep_full(reason):
            meta = dict(full_meta, subset_reason=reason)
            return full_eot, meta
        if original['fsType'] & 0x100:
            optimized.append(keep_full('no_subsetting_license'))
            continue
        try:
            reduced = _build_font_subset(raw, codepoints)
        except (UnsupportedPDF, NodePairingError):
            raise
        except Exception:
            optimized.append(keep_full('subset_error'))
            continue
        if reduced == raw:
            optimized.append(keep_full('no_size_gain'))
            continue
        try:
            _validate_font_subset(raw, reduced, codepoints)
            packed, embedded = _eot_bytes(reduced, is_subset=True)
        except (UnsupportedPDF, NodePairingError):
            raise
        except Exception:
            optimized.append(keep_full('validation_error'))
            continue
        packed_size = _deflated_size(packed)
        if packed_size >= full_meta['source_deflated_bytes']:
            optimized.append(keep_full('no_size_gain'))
            continue
        meta = dict(full_meta)
        meta.update(embedded)
        meta.update(sha256=original['sha256'],
                    embedded_font_sha256=embedded['sha256'],
                    embedding='subset', is_subset=True, subset_reason='used_characters',
                    embedded_deflated_bytes=packed_size)
        optimized.append((packed, meta))
    return optimized
~~~

def _font_subsetting_report(prepared):
    fonts = [meta for _, meta in prepared]
    reasons = {}
    for meta in fonts:
        if meta['embedding'] == 'full':
            reason = meta['subset_reason']
            reasons[reason] = reasons.get(reason, 0) + 1
    return {
        'subset_fonts': sum(m['embedding'] == 'subset' for m in fonts),
        'full_fonts': sum(m['embedding'] == 'full' for m in fonts),
        'source_font_bytes': sum(m['source_font_bytes'] for m in fonts),
        'embedded_font_bytes': sum(m['font_bytes'] for m in fonts),
        'source_deflated_bytes': sum(m['source_deflated_bytes'] for m in fonts),
        'embedded_deflated_bytes': sum(m['embedded_deflated_bytes'] for m in fonts),
        'full_font_reasons': reasons,
