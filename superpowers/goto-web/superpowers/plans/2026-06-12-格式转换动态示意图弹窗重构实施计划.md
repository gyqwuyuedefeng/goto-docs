- [ ] **Step 4: 新增方法**

在 `methods` 中新增：

```js
/**
 * 打开示意图弹窗，默认展示宽转长 Tab，重置图片加载错误状态
 */
openGuideDialog() {
  this.guideActiveTab = 'w2l'
  this.guideImageLoadError = false
  this.guideDialogVisible = true
},
/**
 * 关闭示意图弹窗
 */
closeGuideDialog() {
  this.guideDialogVisible = false
},
/**
 * 切换 Tab 时更新当前激活 Tab 并重置错误状态
 */
onGuideTabChange(tab) {
  this.guideActiveTab = tab.name
  this.guideImageLoadError = false
},
/**
 * 图片加载失败时的回调，设置兜底提示状态
 */
onGuideImageError() {
  this.guideImageLoadError = true
}
```

- [ ] **Step 5: 新增模板 — Icon 入口**

- [ ] **Step 6: 新增模板 — Tab 弹窗**

在 `</template>` 之前、`</Dialog>` 之后（组件根层级）添加：

```html
<!-- 格式转换动态示意图弹窗 -->
<el-dialog
  :visible.sync="guideDialogVisible"
  title="动态示意图"
  custom-class="format-guide-dialog"
  width="800px"
  :close-on-click-modal="true"
  :close-on-press-escape="true"
  :show-close="true"
  @close="closeGuideDialog"
>
  <el-tabs v-model="guideActiveTab" @tab-click="onGuideTabChange">
    <el-tab-pane label="宽转长动态示意图" name="w2l">
      <div class="format-guide-body">
        <img
          v-if="!guideImageLoadError"
          :src="currentGuideImage"
          alt="宽转长动态示意图"
          class="format-guide-image"
          @error="onGuideImageError"
        >
        <div v-else class="format-guide-error">
          示意图加载失败，请稍后重试
        </div>
      </div>
    </el-tab-pane>
    <el-tab-pane label="长转宽动态示意图" name="l2w">
      <div class="format-guide-body">
        <img
          v-if="!guideImageLoadError"
          :src="currentGuideImage"
          alt="长转宽动态示意图"
          class="format-guide-image"
          @error="onGuideImageError"
        >
        <div v-else class="format-guide-error">
          示意图加载失败，请稍后重试
        </div>
      </div>
    </el-tab-pane>
  </el-tabs>
</el-dialog>
```

> ⚠️ `currentGuideImage` 在两个 tab-pane 中都引用同一个计算属性。由于 `el-tabs` 切换时 `guideActiveTab` 已更新，`currentGuideImage` 会自动返回正确的 SVG。`onGuideTabChange` 中会重置 `guideImageLoadError`，确保切换到新 Tab 时不会再显示旧的错误状态。
