/**
 * 为 HTML 交互图构造既有 SVG 转换命令。
 * 桑基图 SVG 的文字使用 pt，节点坐标使用 viewBox 单位；PDF 若沿用 300 DPI，
 * 页面坐标会缩小而文字保持物理字号，故仅此格式按浏览器的 96 DPI 换算。
 * 网络图及其他格式不传 -d，继续遵守转换脚本原有默认值。
 */
String[] buildFileConvertCommand(String inputPath, String outputPath, String imageType) {
    List<String> command = new ArrayList<>(List.of(
            "sh",
            componentBean.getProperties().getConfigProperties().getR().getFileConvertScript(),
            "-i", inputPath,
            "-o", outputPath
    ));
    if (plotType.getGroupId() == 15L && "pdf".equalsIgnoreCase(imageType)) {
        command.add("-d");
        command.add("96");
    }
    return command.toArray(new String[0]);
}

/**
 * 为桑基图 PDF 制作独立转换输入，绕过当前 rsvg-convert 对节点阴影滤镜的解析缺陷。
 * 仅移除节点 rect 上指向 shadow-N 的 filter 属性，不修改原件、其他滤镜或 PNG/SVG 输入。
 * 临时副本放在 ZIP 源目录的父目录，调用者必须在转换后删除返回的不同路径。
 *
 * @param svgPath 页面同步后的原始 SVG 路径
 * @param imageType 当前请求的下载格式
 * @return PDF 专用临时路径，或其他格式的原路径
 * @throws IOException 读取、写入失败；异常时先清理由本方法创建的副本
 */
Path prepareFileConvertInput(Path svgPath, String imageType) throws IOException {
    if (plotType.getGroupId() != 15L || !"pdf".equalsIgnoreCase(imageType)) {
        return svgPath;
    }
    Path temporary = Files.createTempFile(tmpSoftwareFolder.toPath().getParent(),
            tmpFolderName + "-sankey-pdf-", ".svg");
    try {
        String source = Files.readString(svgPath, StandardCharsets.UTF_8);
        String converted = SANKEY_NODE_SHADOW.matcher(source).replaceAll("$1$2");
        Files.writeString(temporary, converted, StandardCharsets.UTF_8);
        return temporary;
    } catch (IOException | RuntimeException e) {
        try {
            Files.deleteIfExists(temporary);
        } catch (IOException cleanupError) {
            e.addSuppressed(cleanupError);
        }
        throw e;
    }
}
