# goto-web 未保存列名标记阻止分析 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在绘图页“开始分析”前阻止未保存的列名标记提交，并提示 `请先保存标记`。

**Architecture:** 在列名标记保存组件中维护局部 dirty 状态，用户确认列名选择后置为未保存，保存成功或重新加载成功后清除。`PlotResultCanvas.vue` 聚合所有文件标记组件的 dirty 状态，`draw.vue` 在现有必选标记校验之前先拦截未保存标记，不改后端接口与保存语义。

**Tech Stack:** Vue 2 SFC, Jest, Vue CLI Unit Test, Element UI, existing `ToastificationNotice`

```js
/* eslint-env jest */

const mockSaveFileMarkSettings = jest.fn()
const mockObtainInitHeaderInfoListAndUploadFileInfo = jest.fn()

jest.mock('@/api/plotType', () => ({
  saveFileMarkSettings: mockSaveFileMarkSettings,

jest.mock('@/constant/modify/publicFunction', () => ({
  filteredHeaderInfoList: jest.fn(() => []),
  obtainCommonParams: jest.fn(() => ({}))
}))
jest.mock('@/constant/associationParamModify', () => ({
  saveFileMarkSuccess: jest.fn()
}))

const plotMixin = require('@/views/goto/plot/type/fileMark/plot').default

function createVm(overrides = {}) {
  const vm = {
    ...plotMixin.data(),
    CommonUtils: {
      deepClone(value) {
        return JSON.parse(JSON.stringify(value))
      },
      isNotEmptyObject(value) {
        return value !== null && value !== undefined
      },
      isNotEmptyArray(value) {
        return Array.isArray(value) && value.length > 0
      }
    },
    ToastificationNotice: {
      notifySuccess: jest.fn(),
      notifyError: jest.fn()
    },
    $set(target, key, value) {
      target[key] = value
    },
    beforeSave: jest.fn(() => true),
    saveFileMarkSuccess: jest.fn(),
    plotType: {
      groupId: 7,
      id: 22
    },
    parent: {
      fileSettings: []
    },
    fileSettingsName: 'fileSettings',
    fileSettingIndex: 0,
    userDrawId: 1001,
    tmpFolderName: 'tmp-folder',
    example: false,
    currentChartUploadFileSettings: {
      name: 'table',
      contentType: 'XLSX',
      uploadFileInfo: {
        accept: '.xlsx'
      },
      headerInfoList: []
    },
    ...overrides
  }

  Object.keys(plotMixin.methods).forEach((methodName) => {
    vm[methodName] = plotMixin.methods[methodName]
  })

  return vm
}

describe('file mark unsaved dirty state', () => {
  beforeEach(() => {
    mockSaveFileMarkSettings.mockReset()
    mockObtainInitHeaderInfoListAndUploadFileInfo.mockReset()
  })

  test('初始状态没有未保存标记', () => {
    const vm = createVm()

    expect(vm.hasUnsavedFileMark()).toBe(false)
  })

  test('确认选择列名后标记为未保存', () => {
    const vm = createVm()

    vm.markFileDirty()

    expect(vm.hasUnsavedFileMark()).toBe(true)
  })

  test('保存成功后清除未保存状态', async () => {
    mockSaveFileMarkSettings.mockResolvedValue({ code: 200 })
    const vm = createVm({
      markDirty: true
    })

    await vm.save()

    expect(mockSaveFileMarkSettings).toHaveBeenCalledTimes(1)
    expect(vm.hasUnsavedFileMark()).toBe(false)
    expect(vm.ToastificationNotice.notifySuccess).toHaveBeenCalledWith(vm, '保存标记成功！')
    expect(vm.saveFileMarkSuccess).toHaveBeenCalledTimes(1)
  })

  test('保存失败后保持未保存状态', async () => {
    mockSaveFileMarkSettings.mockResolvedValue({ code: 500 })
    const vm = createVm({
      markDirty: true
    })

    await vm.save()

    expect(mockSaveFileMarkSettings).toHaveBeenCalledTimes(1)
    expect(vm.hasUnsavedFileMark()).toBe(true)
    expect(vm.ToastificationNotice.notifyError).toHaveBeenCalledWith(vm, '保存标记失败！')
  })

  test('重新加载列名成功后清除未保存状态', async () => {
    const headerInfoList = [{ columnName: 'sample', status: 0 }]
    mockObtainInitHeaderInfoListAndUploadFileInfo.mockResolvedValue({
      code: 200,
      data: [{ headerInfoList }]
    })
    const vm = createVm({
      markDirty: true
    })

    await vm.obtainInitHeaderInfoListAndUploadFileInfo({
      name: 'table',
      contentType: 'XLSX',
      permitContentType: ['XLSX'],
      index: 0
    })

    expect(mockObtainInitHeaderInfoListAndUploadFileInfo).toHaveBeenCalledTimes(1)
    expect(vm.currentChartUploadFileSettings.headerInfoList).toBe(headerInfoList)
    expect(vm.hasUnsavedFileMark()).toBe(false)
  })
})
```
