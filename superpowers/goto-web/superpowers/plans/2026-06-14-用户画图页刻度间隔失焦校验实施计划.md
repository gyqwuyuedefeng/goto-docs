# 用户画图页刻度间隔失焦校验实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 修复用户画图页 `moduleType === 24` 刻度间隔参数的失焦校验，让 `start/end/interval` 在输入过程中不被打断，并在当前字段失焦后按标记列数值范围只修正当前字段。

**Architecture:** 在 `draw/input.vue` 透传失焦事件，在 `draw/scaleIntervalParam.vue` 把失焦字段名向上抛出，在 `drawGroupOptional.vue` 接收事件并调用公共范围函数。`publicFunction.js` 扩展 `syncScaleIntervalRangeByMarkColumns`，支持 `currentSetting` 和 `targetField`，硬校验保持整体处理，轻校验只处理当前字段。

**Tech Stack:** Vue 2、Element UI、Jest、现有 `scripts/run-unit-tests.js` 单元测试入口。

- [ ] **Step 7: 实现当前字段轻校验**

在 `goto-web/src/constant/modify/publicFunction.js` 中替换 `syncScaleIntervalRangeByMarkColumns` 函数为：

```js
/**
 * 根据 scaleFileMarkColumnList 关联标记列的数值范围同步刻度间隔参数
 *
 * @param {Object} commonParams - 公共参数对象
 * @param {Object} options - 选项
 * @param {boolean} [options.hard=false] - 是否硬校验模式
 * @param {String} [options.targetField] - 轻校验目标字段：start、end、interval
 */
function syncScaleIntervalRangeByMarkColumns(commonParams, options) {
  const hard = options && options.hard === true
  const targetField = options && options.targetField

  if (!commonParams) {
    return
  }

  const settings = resolveScaleIntervalSettings(commonParams)
  if (!settings.length) {
    return
  }

  for (const setting of settings) {
    syncSingleScaleIntervalSetting(commonParams, setting, { hard, targetField })
  }
}

/**
 * 解析需要校验的刻度间隔参数。
 * 失焦校验会显式传入 currentSetting，硬校验仍兼容扫描顶层和分组内参数。
 * @param {Object} commonParams - 公共参数对象
 * @returns {Array<Object>} 待校验的 moduleType 24 参数列表
 */
function resolveScaleIntervalSettings(commonParams) {
  if (commonParams.currentSetting && commonParams.currentSetting.moduleType === 24) {
    return [commonParams.currentSetting]
  }

  const result = []
  collectScaleIntervalSettings(commonParams.topParamSettingsList, result)
  return result
}

/**
 * 递归收集 moduleType 24 参数，兼容参数分组结构。
 * @param {Array} list - 参数列表
 * @param {Array} result - 收集结果
 */
function collectScaleIntervalSettings(list, result) {
  if (!Array.isArray(list)) {
    return
  }

  for (const setting of list) {
    if (!setting) {
      continue
    }
    if (setting.moduleType === 24) {
      result.push(setting)
    }
    if (Array.isArray(setting.paramSettingsList)) {
      collectScaleIntervalSettings(setting.paramSettingsList, result)
    }
  }
}

/**
 * 同步单个刻度间隔参数。
 * @param {Object} commonParams - 公共参数对象
 * @param {Object} setting - moduleType 24 参数
 * @param {Object} options - 校验选项
 */
function syncSingleScaleIntervalSetting(commonParams, setting, options) {
  const markColumnList = setting.scaleFileMarkColumnList
  if (!markColumnList || markColumnList.length === 0) {
    return
  }

  const range = calculateScaleIntervalRange(commonParams, markColumnList)
  if (!range) {
    if (options.hard) {
      clearScaleIntervalFields(setting, commonParams.vueContext)
    }
    return
  }

  if (options.hard) {
    syncScaleIntervalHard(commonParams, setting, range)
    return
  }

  syncScaleIntervalTargetField(commonParams, setting, range, options.targetField)
}
```

在 `clearScaleIntervalFields` 前增加以下辅助函数：

```js
/**
 * 对刻度间隔参数执行硬校验，整体处理 start/end/interval。
 * @param {Object} commonParams - 公共参数对象
 * @param {Object} setting - moduleType 24 参数
 * @param {Object} range - 数值范围
 */
function syncScaleIntervalHard(commonParams, setting, range) {
  const hasStart = hasScaleIntervalValue(setting.start)
  const hasEnd = hasScaleIntervalValue(setting.end)
  const hasInterval = hasScaleIntervalValue(setting.interval)

  if (!hasStart && !hasEnd && !hasInterval) {
    return
  }

  if (!hasStart || !hasEnd || !hasInterval) {
    clearScaleIntervalFields(setting, commonParams.vueContext)
    return
  }

  syncScaleIntervalField(commonParams, setting, range, 'start')
  syncScaleIntervalField(commonParams, setting, range, 'end')
  syncScaleIntervalField(commonParams, setting, range, 'interval')
}

/**
 * 对当前失焦字段执行轻校验。
 * 输入过程中不调用该方法，只有 blur/change 完成编辑后调用。
 * @param {Object} commonParams - 公共参数对象
 * @param {Object} setting - moduleType 24 参数
 * @param {Object} range - 数值范围
 * @param {String} targetField - 当前失焦字段
 */
function syncScaleIntervalTargetField(commonParams, setting, range, targetField) {
  if (!['start', 'end', 'interval'].includes(targetField)) {
    return
  }
  if (!hasScaleIntervalValue(setting[targetField])) {
    return
  }
  syncScaleIntervalField(commonParams, setting, range, targetField)
}

/**
 * 校验并修正单个刻度间隔字段。
 * @param {Object} commonParams - 公共参数对象
 * @param {Object} setting - moduleType 24 参数
 * @param {Object} range - 数值范围
 * @param {String} fieldName - 字段名
 */
function syncScaleIntervalField(commonParams, setting, range, fieldName) {
  const { min, max, span } = range
  const lowerBound = min - span
  const upperBound = max + span

  if (fieldName === 'start') {
    let nextStart = Number(setting.start)
    if (Number.isNaN(nextStart)) {
      return
    }
    if (nextStart < lowerBound) {
      nextStart = lowerBound
    }
    if (hasScaleIntervalValue(setting.end)) {
      const end = Number(setting.end)
      if (!Number.isNaN(end) && nextStart >= end) {
        nextStart = end - span
      }
    }
    setScaleIntervalField(setting, 'start', nextStart, commonParams.vueContext)
    return
  }

  if (fieldName === 'end') {
    let nextEnd = Number(setting.end)
    if (Number.isNaN(nextEnd)) {
      return
    }
    if (nextEnd > upperBound) {
      nextEnd = upperBound
    }
    if (hasScaleIntervalValue(setting.start)) {
      const start = Number(setting.start)
      if (!Number.isNaN(start) && nextEnd <= start) {
        nextEnd = start + span
      }
    }
    setScaleIntervalField(setting, 'end', nextEnd, commonParams.vueContext)
    return
  }

  if (fieldName === 'interval') {
    let nextInterval = Number(setting.interval)
    if (Number.isNaN(nextInterval)) {
      return
    }
    if (nextInterval <= 0 || nextInterval > span) {
      nextInterval = span
    }
    setScaleIntervalField(setting, 'interval', nextInterval, commonParams.vueContext)
  }
}

/**
 * 判断刻度间隔字段是否已填写。
 * @param {*} value - 字段值
 * @returns {boolean} 是否有值
 */
function hasScaleIntervalValue(value) {
  return value !== null && value !== undefined && value !== ''
}

/**
 * 设置刻度间隔字段，优先走 Vue 响应式更新。
 * @param {Object} setting - moduleType 24 参数
 * @param {String} fieldName - 字段名
 * @param {Number} value - 新值
 * @param {Object} vueContext - Vue 实例
 */
function setScaleIntervalField(setting, fieldName, value, vueContext) {
  if (vueContext) {
    vueContext.$set(setting, fieldName, value)
  } else {
    setting[fieldName] = value
  }
}
```
