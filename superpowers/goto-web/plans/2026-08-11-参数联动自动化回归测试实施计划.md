创建 `associationTestBuilders.js`：

```js
function cloneFixture(value) {
  if (value === undefined) return undefined
  return JSON.parse(JSON.stringify(value))
}

function createVueContext() {
  return {
    $set: jest.fn((target, key, value) => {
      target[key] = value
    }),
    $delete: jest.fn((target, key) => {
      delete target[key]
    }),
    $forceUpdate: jest.fn()
  }
}

function createSetting(paramName, overrides = {}) {
  return {
    moduleType: 3,
    paramName,
    value: null,
    visible: true,
    ...cloneFixture(overrides)
  }
}

function createGroup(name, settings = [], overrides = {}) {
  return {
    moduleType: -1,
    name,
    visible: true,
    paramSettingsList: cloneFixture(settings),
    ...cloneFixture(overrides)
  }
}

function createFileSetting(overrides = {}) {
  const source = cloneFixture(overrides)
  const header = source.header || []
  const body = source.body || []
  delete source.header
  delete source.body
  return {
    name: 'input.xlsx',
    uploadFileInfo: {
      uploaded: true,
      map: {},
      handsontable: { header, body }
    },
    headerInfoList: [],
    fileMarkContainerList: [],
    ...source
  }
}

function findSetting(commonParams, paramName) {
  for (const group of commonParams.topParamSettingsList || []) {
    for (const setting of group.paramSettingsList || []) {
      if (setting.paramName === paramName) return setting
    }
  }
  return null
}

function findGroup(commonParams, groupName) {
  return (commonParams.topParamSettingsList || []).find(
    group => group.moduleType === -1 && group.name === groupName
  ) || null
}

function createCommonParams(options = {}) {
  const scenarioName = options.scenarioName || '未命名场景'
  const commonParams = {
    vueContext: createVueContext(),
    setting: null,
    fileSettings: cloneFixture(options.fileSettings || []),
    groupId: options.groupId,
    plotTypeId: options.plotTypeId || 1,
    userDrawId: options.userDrawId || 'association-contract',
    topParamSettingsList: cloneFixture(options.topParamSettingsList || []),
    params: cloneFixture(options.params),
    fileSettingIndex: options.fileSettingIndex
  }
  if (options.settingName) {
    commonParams.setting = findSetting(commonParams, options.settingName)
    if (!commonParams.setting) {
      throw new Error(`${scenarioName}: 找不到参数 ${options.settingName}`)
    }
  }
  return commonParams
}

function projectSettings(commonParams, selectors) {
  const result = {}
  Object.keys(selectors).forEach(paramName => {
    const setting = findSetting(commonParams, paramName)
    if (!setting) throw new Error(`状态投影找不到参数 ${paramName}`)
    result[paramName] = {}
    selectors[paramName].forEach(key => {
      result[paramName][key] = cloneFixture(setting[key])
    })
  })
  return result
}

module.exports = {
  cloneFixture,
  createVueContext,
  createSetting,
  createGroup,
  createFileSetting,
  findSetting,
  findGroup,
  createCommonParams,
  projectSettings
}
```

const { createCommonParams } = require('./associationTestBuilders')
