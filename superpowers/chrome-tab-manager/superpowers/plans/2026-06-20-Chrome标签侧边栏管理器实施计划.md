# Chrome 标签侧边栏管理器 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 构建一个本地自用的 Manifest V3 Chrome 插件，用 Side Panel 管理分组网页，并在 Chrome 主标签页中打开真实网页。

**Architecture:** 插件采用无构建步骤的原生 JavaScript 模块结构，便于通过 `chrome://extensions/` 直接加载 `chrome-tab-manager/` 目录。业务逻辑拆分为 `storage`、`tabs`、`importExport`、`contextMenus` 和 `sidepanel`，其中纯逻辑模块使用 Node 内置测试框架和 mock Chrome API 验证，浏览器特性通过手工验收验证。

**Tech Stack:** Chrome Extension Manifest V3、Chrome Side Panel API、`chrome.storage.local`、`chrome.tabs`、`chrome.tabGroups`、`chrome.contextMenus`、原生 HTML/CSS/JavaScript、Node.js `node:test`。

```js
const BLOCKED_PROTOCOLS = ['chrome:', 'chrome-extension:', 'edge:', 'about:', 'devtools:'];

export function createEmptyState() {
  return {
    groups: [],
    meta: {
      lastSkippedDuplicates: 0
    }
  };
}

export function makeId(prefix) {
  return `${prefix}_${Date.now()}_${Math.random().toString(36).slice(2, 8)}`;
}

export function nowIso() {
  return new Date().toISOString();
}

export function createGroup(name, options = {}) {
  const now = options.now || nowIso();
  return {
    id: options.id || makeId('group'),
    name: String(name || '').trim(),
    sortOrder: options.sortOrder || 1,
    expanded: options.expanded ?? true,
    createdAt: options.createdAt || now,
    updatedAt: options.updatedAt || now,
    items: options.items || []
  };
}

export function createItem(input, options = {}) {
  const now = options.now || nowIso();
  return {
    id: options.id || makeId('item'),
    title: String(input.title || input.url || '未命名网页').trim(),
    url: String(input.url || '').trim(),
    source: input.source || 'page',
    sortOrder: options.sortOrder || 1,
    createdAt: options.createdAt || now
  };
}

export function normalizeState(rawState) {
  const state = rawState && Array.isArray(rawState.groups) ? rawState : createEmptyState();
  const groups = state.groups
    .map((group, groupIndex) => ({
      ...createGroup(group.name || '未命名分组', {
        id: group.id || makeId('group'),
        sortOrder: Number(group.sortOrder || groupIndex + 1),
        expanded: group.expanded ?? true,
        createdAt: group.createdAt,
        updatedAt: group.updatedAt,
        items: Array.isArray(group.items) ? group.items : []
      }),
      items: (Array.isArray(group.items) ? group.items : [])
        .filter((item) => item && isSavableUrl(item.url))
        .map((item, itemIndex) => createItem(item, {
          id: item.id || makeId('item'),
          sortOrder: Number(item.sortOrder || itemIndex + 1),
          createdAt: item.createdAt
        }))
        .sort((a, b) => a.sortOrder - b.sortOrder)
    }))
    .sort((a, b) => a.sortOrder - b.sortOrder);

  return {
    groups,
    meta: {
      lastSkippedDuplicates: Number(state.meta?.lastSkippedDuplicates || 0)
    }
  };
}

export function isSavableUrl(url) {
  if (!url || typeof url !== 'string') return false;
  try {
    const parsed = new URL(url);
    return !BLOCKED_PROTOCOLS.includes(parsed.protocol);
  } catch {
    return false;
  }
}

export function addGroup(state, group) {
  const normalized = normalizeState(state);
  const nextGroup = {
    ...group,
    sortOrder: group.sortOrder || normalized.groups.length + 1,
    items: group.items || []
  };
  return normalizeState({
    ...normalized,
    groups: [...normalized.groups, nextGroup]
  });
}

export function updateGroup(state, groupId, patch) {
  const normalized = normalizeState(state);
  return normalizeState({
    ...normalized,
    groups: normalized.groups.map((group) => {
      if (group.id !== groupId) return group;
      return {
        ...group,
        ...patch,
        updatedAt: nowIso()
      };
    })
  });
}

export function deleteGroup(state, groupId) {
  const normalized = normalizeState(state);
  return normalizeState({
    ...normalized,
    groups: normalized.groups.filter((group) => group.id !== groupId)
  });
}

export function addItemToGroup(state, groupId, item) {
  const normalized = normalizeState(state);
  let skipped = 0;
  const groups = normalized.groups.map((group) => {
    if (group.id !== groupId) return group;
    if (!isSavableUrl(item.url) || group.items.some((existing) => existing.url === item.url)) {
      skipped += 1;
      return group;
    }
    return {
      ...group,
      updatedAt: nowIso(),
      items: [
        ...group.items,
        {
          ...item,
          sortOrder: item.sortOrder || group.items.length + 1
        }
      ]
    };
  });
  return normalizeState({
    groups,
    meta: {
      lastSkippedDuplicates: skipped
    }
  });
}

export function deleteItem(state, groupId, itemId) {
  const normalized = normalizeState(state);
  return normalizeState({
    ...normalized,
    groups: normalized.groups.map((group) => {
      if (group.id !== groupId) return group;
      return {
        ...group,
        updatedAt: nowIso(),
        items: group.items.filter((item) => item.id !== itemId)
      };
    })
  });
}

export function reorderGroups(state, orderedGroupIds) {
  const normalized = normalizeState(state);
  const orderMap = new Map(orderedGroupIds.map((id, index) => [id, index + 1]));
  return normalizeState({
    ...normalized,
    groups: normalized.groups.map((group) => ({
      ...group,
      sortOrder: orderMap.get(group.id) || group.sortOrder
    }))
  });
}

export function reorderItems(state, groupId, orderedItemIds) {
  const normalized = normalizeState(state);
  const orderMap = new Map(orderedItemIds.map((id, index) => [id, index + 1]));
  return normalizeState({
    ...normalized,
    groups: normalized.groups.map((group) => {
      if (group.id !== groupId) return group;
      return {
        ...group,
        items: group.items.map((item) => ({
          ...item,
          sortOrder: orderMap.get(item.id) || item.sortOrder
        }))
      };
    })
  });
}
```

```js
export function createChromeMock(initialStorage = {}) {
  const localData = { ...initialStorage };
  const calls = [];

  return {
    calls,
    storage: {
      local: {
        async get(keys) {
          calls.push(['storage.local.get', keys]);
          if (Array.isArray(keys)) {
            return Object.fromEntries(keys.map((key) => [key, localData[key]]));
          }
          if (typeof keys === 'string') {
            return { [keys]: localData[keys] };
          }
          return { ...localData };
        },
        async set(values) {
          calls.push(['storage.local.set', values]);
          Object.assign(localData, values);
        }
      }
    },
    tabs: {
      async query() {
        return [];
      }
    },
    tabGroups: {
      async get(groupId) {
        return { id: groupId, title: `标签组 ${groupId}` };
      }
    }
  };
}
```

```js
import {
  addGroup,
  addItemToGroup,
  createGroup,
  createItem,
  normalizeState
} from './schema.js';

const STORAGE_KEY = 'tabManagerState';

export async function getState(chromeApi = chrome) {
  const data = await chromeApi.storage.local.get([STORAGE_KEY]);
  return normalizeState(data[STORAGE_KEY]);
}

export async function setState(chromeApi = chrome, state) {
  const normalized = normalizeState(state);
  await chromeApi.storage.local.set({ [STORAGE_KEY]: normalized });
  return normalized;
}

export async function createGroupWithName(chromeApi, name) {
  const state = await getState(chromeApi);
  const group = createGroup(name, { sortOrder: state.groups.length + 1 });
  const nextState = addGroup(state, group);
  await setState(chromeApi, nextState);
  return group;
}

export async function saveItemToGroup(chromeApi, groupId, input) {
  const state = await getState(chromeApi);
  const group = state.groups.find((candidate) => candidate.id === groupId);
  const item = createItem(input, { sortOrder: group ? group.items.length + 1 : 1 });
  const nextState = addItemToGroup(state, groupId, item);
  await setState(chromeApi, nextState);
  return {
    item,
    skippedDuplicates: nextState.meta.lastSkippedDuplicates
  };
}

export async function mergeImportedState(chromeApi, importedState) {
  let state = await getState(chromeApi);
  const normalizedImport = normalizeState(importedState);

  for (const importedGroup of normalizedImport.groups) {
    let targetGroup = state.groups.find((group) => group.name === importedGroup.name);
    if (!targetGroup) {
      targetGroup = createGroup(importedGroup.name, { sortOrder: state.groups.length + 1 });
      state = addGroup(state, targetGroup);
    }
    for (const importedItem of importedGroup.items) {
      state = addItemToGroup(state, targetGroup.id, createItem(importedItem, {
        sortOrder: targetGroup.items.length + 1,
        createdAt: importedItem.createdAt
      }));
      targetGroup = state.groups.find((group) => group.id === targetGroup.id);
    }
  }

  return setState(chromeApi, state);
}

export async function overwriteState(chromeApi, importedState) {
  return setState(chromeApi, importedState);
}
```

```js
import { isSavableUrl, normalizeState } from './schema.js';

export function exportStateToJson(state) {
  return JSON.stringify(normalizeState(state), null, 2);
}

export function parseImportedJson(jsonText) {
  let parsed;
  try {
    parsed = JSON.parse(jsonText);
  } catch {
    throw new Error('导入文件不是合法 JSON');
  }
  validateImportedState(parsed);
  return normalizeState(parsed);
}

export function validateImportedState(state) {
  if (!state || !Array.isArray(state.groups)) {
    throw new Error('导入文件缺少 groups 数组');
  }

  for (const group of state.groups) {
    if (!group || typeof group.name !== 'string') {
      throw new Error('导入文件包含无效分组');
    }
    if (group.items !== undefined && !Array.isArray(group.items)) {
      throw new Error('导入文件包含无效网页列表');
    }
    for (const item of group.items || []) {
      if (!item || typeof item.url !== 'string') {
        throw new Error('导入文件包含无效网页');
      }
      if (!isSavableUrl(item.url)) {
        throw new Error(`导入文件包含不可保存的 URL：${item.url}`);
      }
    }
  }
}
```

```js
import { isSavableUrl } from '../shared/schema.js';

export async function openOrFocusUrl(chromeApi = chrome, url) {
  const existingTabs = await chromeApi.tabs.query({ currentWindow: true, url });
  if (existingTabs.length > 0) {
    const tab = await chromeApi.tabs.update(existingTabs[0].id, { active: true });
    return { action: 'focused', tab };
  }

  const tab = await chromeApi.tabs.create({ url, active: true });
  return { action: 'created', tab };
}

export async function restoreGroupToCurrentWindow(chromeApi = chrome, group) {
  const tabIds = [];
  for (const item of group.items || []) {
    if (!isSavableUrl(item.url)) continue;
    const tab = await chromeApi.tabs.create({ url: item.url, active: false });
    tabIds.push(tab.id);
  }
  await groupCreatedTabs(chromeApi, tabIds, group.name);
  return { openedCount: tabIds.length };
}

export async function restoreGroupToNewWindow(chromeApi = chrome, group) {
  const createdWindow = await chromeApi.windows.create({ focused: true });
  const tabIds = [];
  for (const item of group.items || []) {
    if (!isSavableUrl(item.url)) continue;
    const tab = await chromeApi.tabs.create({ windowId: createdWindow.id, url: item.url, active: false });
    tabIds.push(tab.id);
  }
  await groupCreatedTabs(chromeApi, tabIds, group.name);
  return { openedCount: tabIds.length, windowId: createdWindow.id };
}

async function groupCreatedTabs(chromeApi, tabIds, title) {
  if (tabIds.length === 0) return null;
  const groupId = await chromeApi.tabs.group({ tabIds });
  await chromeApi.tabGroups.update(groupId, { title, color: 'blue' });
  return groupId;
}
```

```js
      id: `${LINK_PREFIX}${group.id}`,
      parentId: ROOT_MENU_ID,

export async function handleContextMenuClick(chromeApi = chrome, info, tab) {
  if (typeof info.menuItemId !== 'string') return null;

  if (info.menuItemId.startsWith(PAGE_PREFIX)) {
    const groupId = info.menuItemId.slice(PAGE_PREFIX.length);
    return saveItemToGroup(chromeApi, groupId, {
      title: tab?.title || tab?.url || '未命名网页',
      url: tab?.url,
      source: 'page'
    });
  }

  if (info.menuItemId.startsWith(LINK_PREFIX)) {
    const groupId = info.menuItemId.slice(LINK_PREFIX.length);
    return saveItemToGroup(chromeApi, groupId, {
      title: info.linkUrl,
      url: info.linkUrl,
      source: 'link'
```

```js
async function saveCurrentPage() {
  const [tab] = await chrome.tabs.query({ active: true, currentWindow: true });
  const group = await promptGroup('保存当前页到分组');
  if (!group || !tab?.url) return;
  await saveItemToGroup(chrome, group.id, { title: tab.title, url: tab.url, source: 'page' });
  await refresh('已保存当前页');
}

async function saveCurrentWindow() {
  const group = await promptGroup('保存当前窗口到分组');
  if (!group) return;
  const tabs = await chrome.tabs.query({ currentWindow: true });
  for (const tab of tabs) {
    await saveItemToGroup(chrome, group.id, { title: tab.title, url: tab.url, source: 'window' });
  }
  await refresh('已保存当前窗口');
}

async function saveCurrentTabGroups() {
  const tabs = await chrome.tabs.query({ currentWindow: true });
  const grouped = new Map();
  for (const tab of tabs.filter((candidate) => candidate.groupId !== undefined && candidate.groupId >= 0)) {
    if (!grouped.has(tab.groupId)) grouped.set(tab.groupId, []);
    grouped.get(tab.groupId).push(tab);
  }
  for (const [groupId, groupTabs] of grouped) {
    const chromeGroup = await chrome.tabGroups.get(groupId);
    const group = await createGroupWithName(chrome, chromeGroup.title || '未命名标签组');
    for (const tab of groupTabs) {
      await saveItemToGroup(chrome, group.id, { title: tab.title, url: tab.url, source: 'tabGroup' });
    }
  }
  await refresh('已保存当前标签组');
}

async function promptGroup(title) {
  const name = window.prompt(`${title}\n请输入新分组名称，或输入已有分组名称：`);
  if (!name) return null;
  const existing = currentState.groups.find((group) => group.name === name.trim());
  if (existing) return existing;
  return createGroupWithName(chrome, name.trim());
}

async function deleteGroupAfterConfirm(group) {
  if (!window.confirm(`将删除分组“${group.name}”及其下 ${group.items.length} 个网页，此操作不会关闭已经打开的 Chrome 标签页。确认删除？`)) return;
  currentState = deleteGroup(currentState, group.id);
  await setState(chrome, currentState);
  render();
  showToast('已删除分组');
}

async function deleteItemAfterConfirm(group, item) {
  if (!window.confirm(`确认删除网页“${item.title}”？此操作不会关闭已经打开的 Chrome 标签页。`)) return;
  currentState = deleteItem(currentState, group.id, item.id);
  await setState(chrome, currentState);
  render();
  showToast('已删除网页');
}

function attachGroupDrag(element) {
  element.addEventListener('dragstart', () => element.classList.add('dragging'));
  element.addEventListener('dragend', async () => {
    element.classList.remove('dragging');
    const orderedIds = [...treeRoot.querySelectorAll('.group')].map((node) => node.dataset.groupId);
    currentState = reorderGroups(currentState, orderedIds);
    await setState(chrome, currentState);
    render();
  });
  element.addEventListener('dragover', (event) => {
    event.preventDefault();
    const dragging = treeRoot.querySelector('.group.dragging');
    if (dragging && dragging !== element) {
      treeRoot.insertBefore(dragging, element);
    }
  });
}

function attachItemDrag(element) {
  element.addEventListener('dragstart', () => element.classList.add('dragging'));
  element.addEventListener('dragend', async () => {
    element.classList.remove('dragging');
    const groupId = element.dataset.groupId;
    const container = treeRoot.querySelector(`.items[data-group-id="${groupId}"]`);
    const orderedIds = [...container.querySelectorAll('.item-row')].map((node) => node.dataset.itemId);
    currentState = reorderItems(currentState, groupId, orderedIds);
    await setState(chrome, currentState);
    render();
  });
  element.addEventListener('dragover', (event) => {
    event.preventDefault();
    const container = element.closest('.items');
    const dragging = container.querySelector('.item-row.dragging');
    if (dragging && dragging !== element) {
      container.insertBefore(dragging, element);
    }
  });
}

function exportJson() {
  const blob = new Blob([exportStateToJson(currentState)], { type: 'application/json' });
  const url = URL.createObjectURL(blob);
  const link = document.createElement('a');
  link.href = url;
  link.download = `chrome-tab-manager-${new Date().toISOString().slice(0, 10)}.json`;
  link.click();
  URL.revokeObjectURL(url);
}

async function importJson(event) {
  const file = event.target.files?.[0];
  if (!file) return;
  try {
    const imported = parseImportedJson(await file.text());
    if (window.confirm('选择“确定”覆盖导入，选择“取消”合并导入。')) {
      currentState = imported;
      await setState(chrome, currentState);
    } else {
      const { mergeImportedState } = await import('../shared/storage.js');
      currentState = await mergeImportedState(chrome, imported);
    }
    render();
    showToast('导入完成');
  } catch (error) {
    showToast(error.message);
  } finally {
    event.target.value = '';
  }
}

async function refresh(message) {
  currentState = await getState(chrome);
  render();
  showToast(message);
}

function showToast(message) {
  toast.textContent = message;
  toast.hidden = false;
  window.clearTimeout(showToast.timer);
  showToast.timer = window.setTimeout(() => {
    toast.hidden = true;
  }, 2200);
}

function escapeHtml(value) {
  return String(value).replace(/[&<>"']/g, (char) => ({
    '&': '&amp;',
    '<': '&lt;',
    '>': '&gt;',
    '"': '&quot;',
    "'": '&#39;'
  })[char]);
}
```
