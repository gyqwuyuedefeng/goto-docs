重写 `src/shared/schema.js`。保留现有工具函数（`generateId`、`generateTimestamp`、`getDomainFromUrl`、`getFaviconUrl`、`isSavableUrl`、`createDefaultUi`、`clampNavWidth`、`UI_DEFAULTS`、`NAV_WIDTH`、`updateUiPreferences`）。修改和新增以下函数：

```js
// --- 分组与网页创建 ---

/**
 * 创建新分组。
 * 默认 parentId 为 null（根级），children 为空数组，不再有独立的 items 字段。
 *
 * @param {string} name - 分组名称
 * @param {Object} [overrides] - 可选的字段覆盖
 * @returns {Object} 新分组对象
 */
export function createGroup(name, overrides = {}) {
  return {
    id: overrides.id || generateId(),
    name,
    parentId: overrides.parentId !== undefined ? overrides.parentId : null,
    sortOrder: overrides.sortOrder != null ? overrides.sortOrder : 0,
    expanded: overrides.expanded !== undefined ? overrides.expanded : true,
    pinned: overrides.pinned === true,
    createdAt: overrides.createdAt || generateTimestamp(),
    updatedAt: overrides.updatedAt || overrides.createdAt || generateTimestamp(),
    children: Array.isArray(overrides.children) ? overrides.children : []
  };
}

/**
 * 创建网页项。
 * 不再包含 sortOrder 字段，排序由父分组 children 数组位置决定。
 *
 * @param {Object} input - 网页基本信息 { title, url, source }
 * @param {Object} [overrides] - 可选的字段覆盖
 * @returns {Object} 网页项对象
 */
export function createItem(input, overrides = {}) {
  return {
    id: overrides.id || generateId(),
    title: input.title || input.url || '未命名网页',
    url: input.url || '',
    source: input.source || 'page',
    favorite: overrides.favorite === true,
    faviconUrl: overrides.faviconUrl || getFaviconUrl(input.url || ''),
    createdAt: overrides.createdAt || generateTimestamp()
  };
}

// --- 数据规范化与迁移 ---

/**
 * 规范化状态对象，旧数据自动迁移为 children + parentId 格式。
 * 旧 items 数组中的网页转换为 children 中的 kind:'item' 条目。
 *
 * @param {Object} [state] - 原始状态对象
 * @returns {Object} 规范化后的状态
 */
export function normalizeState(state) {
  const raw = state && typeof state === 'object' ? state : {};
  const groups = Array.isArray(raw.groups) ? raw.groups : [];

  return {
    ui: createDefaultUi(raw.ui || {}),
    groups: groups
      .filter((g) => g && typeof g.id === 'string')
      .map((g) => {
        // 旧 items 迁移为 children
        let children;
        if (Array.isArray(g.children)) {
          children = g.children;
        } else if (Array.isArray(g.items)) {
          children = g.items
            .filter((it) => it && typeof it.id === 'string')
            .map((it) => ({
              kind: 'item',
              id: it.id,
              title: typeof it.title === 'string' ? it.title : it.url || '未命名网页',
              url: typeof it.url === 'string' ? it.url : '',
              source: typeof it.source === 'string' ? it.source : 'page',
              favorite: it.favorite === true,
              faviconUrl: typeof it.faviconUrl === 'string' && it.faviconUrl
                ? it.faviconUrl
                : getFaviconUrl(it.url || ''),
              createdAt: it.createdAt || generateTimestamp()
            }));
        } else {
          children = [];
        }

        return {
          id: g.id,
          name: typeof g.name === 'string' ? g.name : '未命名分组',
          parentId: g.parentId || null,
          sortOrder: Number.isFinite(g.sortOrder) ? g.sortOrder : 0,
          expanded: typeof g.expanded === 'boolean' ? g.expanded : true,
          pinned: g.pinned === true,
          createdAt: g.createdAt || generateTimestamp(),
          updatedAt: g.updatedAt || g.createdAt || generateTimestamp(),
          children
        };
      })
  };
}

// --- 分组查找 ---

/**
 * 在 groups 数组中按 id 查找分组。
 *
 * @param {Object[]} groups - 分组数组
 * @param {string} id - 分组 ID
 * @returns {Object|undefined} 找到的分组或 undefined
 */
export function findGroup(groups, id) {
  return groups.find((g) => g.id === id);
}

/**
 * 检测 targetId 是否是 ancestorId 的后代（含自身）。
 * 递归沿 parentId 链向上查找，避免拖拽产生循环引用。
 *
 * @param {Object[]} groups - 分组数组
 * @param {string} ancestorId - 候选祖先 ID
 * @param {string} targetId - 候选后代 ID
 * @returns {boolean} targetId 是 ancestorId 的后代或自身时返回 true
 */
export function isDescendant(groups, ancestorId, targetId) {
  if (ancestorId === targetId) return true;
  const target = findGroup(groups, targetId);
  if (!target || !target.parentId) return false;
  return isDescendant(groups, ancestorId, target.parentId);
}

/**
 * 递归收集分组及其所有后代中的网页 item id。
 * 用于"恢复分组"时获取所有需打开的网页。
 *
 * @param {Object[]} groups - 分组数组
 * @param {string} groupId - 起始分组 ID
 * @returns {string[]} 所有后代网页的 item id 数组
 */
export function getAllDescendantItemIds(groups, groupId) {
  const group = findGroup(groups, groupId);
  if (!group) return [];
  const ids = [];
  for (const child of group.children) {
    if (child.kind === 'item') {
      ids.push(child.id);
    } else if (child.kind === 'group') {
      ids.push(...getAllDescendantItemIds(groups, child.groupId));
    }
  }
  return ids;
}

// --- children 操作 ---

/**
 * 向分组的 children 末尾追加子内容。
 *
 * @param {Object} state - 完整状态
 * @param {string} groupId - 目标分组 ID
 * @param {Object} child - 子内容条目 { kind: 'item', ... } 或 { kind: 'group', groupId }
 * @returns {Object} 更新后的状态
 */
export function addChildToGroup(state, groupId, child) {
  return {
    ...state,
    groups: state.groups.map((g) =>
      g.id === groupId ? { ...g, children: [...g.children, child], updatedAt: generateTimestamp() } : g
    )
  };
}

/**
 * 从分组的 children 中按索引删除子内容。
 *
 * @param {Object} state - 完整状态
 * @param {string} groupId - 分组 ID
 * @param {number} index - children 中的索引
 * @returns {Object} 更新后的状态
 */
export function removeChildFromGroup(state, groupId, index) {
  return {
    ...state,
    groups: state.groups.map((g) =>
      g.id === groupId
        ? { ...g, children: [...g.children.slice(0, index), ...g.children.slice(index + 1)], updatedAt: generateTimestamp() }
        : g
    )
  };
}

/**
 * 将子内容从源分组移动到目标分组的指定位置。
 * 先在源分组删除，再在目标分组插入。源和目标可以是同一分组。
 *
 * @param {Object} state - 完整状态
 * @param {string} fromGroupId - 源分组 ID
 * @param {number} fromIndex - 源分组 children 索引
 * @param {string} toGroupId - 目标分组 ID
 * @param {number} toIndex - 目标分组 children 索引
 * @returns {Object} 更新后的状态
 */
export function moveChild(state, fromGroupId, fromIndex, toGroupId, toIndex) {
  const child = state.groups.find((g) => g.id === fromGroupId)?.children[fromIndex];
  if (!child) return state;
  // 先在源分组删除
  let s = removeChildFromGroup(state, fromGroupId, fromIndex);
  // 若同分组且删除位置在目标位置前，目标索引需减一
  let adjustedIndex = toIndex;
  if (fromGroupId === toGroupId && fromIndex < toIndex) {
    adjustedIndex--;
  }
  // 在目标分组插入
  return {
    ...s,
    groups: s.groups.map((g) =>
      g.id === toGroupId
        ? { ...g, children: [...g.children.slice(0, adjustedIndex), child, ...g.children.slice(adjustedIndex)], updatedAt: generateTimestamp() }
        : g
    )
  };
}

// --- 兼容旧接口（保留但适配 children） ---

/**
 * 添加分组到状态（兼容旧接口）。
 */
export function addGroup(state, group) {
  if (state.groups.some((g) => g.id === group.id)) return state;
  return { ...state, groups: [...state.groups, { ...group, children: group.children || [] }] };
}

/**
 * 删除分组同时从父分组的 children 中移除子分组引用。
 */
export function deleteGroup(state, groupId) {
  const toDelete = findGroup(state.groups, groupId);
  let s = { ...state, groups: state.groups.filter((g) => g.id !== groupId) };
  // 从父分组的 children 中移除引用
  if (toDelete && toDelete.parentId) {
    s = {
      ...s,
      groups: s.groups.map((g) => {
        if (g.id !== toDelete.parentId) return g;
        return {
          ...g,
          children: g.children.filter((c) => !(c.kind === 'group' && c.groupId === groupId)),
          updatedAt: generateTimestamp()
        };
      })
    };
  }
  return s;
}

/**
 * 更新分组字段。
 */
export function updateGroup(state, groupId, patches) {
  return {
    ...state,
    groups: state.groups.map((g) =>
      g.id === groupId ? { ...g, ...patches, updatedAt: generateTimestamp() } : g
    )
  };
}

/**
 * 切换置顶。
 */
export function toggleGroupPinned(state, groupId) {
  return updateGroup(state, groupId, { pinned: !findGroup(state.groups, groupId)?.pinned });
}

/**
 * 在 children 中切换网页常用。
 */
export function toggleItemFavorite(state, groupId, itemId) {
  return {
    ...state,
    groups: state.groups.map((g) => {
      if (g.id !== groupId) return g;
      return {
        ...g,
        children: g.children.map((c) =>
          c.kind === 'item' && c.id === itemId ? { ...c, favorite: !c.favorite } : c
        ),
        updatedAt: generateTimestamp()
      };
    })
  };
}

/**
 * 展示用排序：置顶分组优先，再按 sortOrder。
 */
export function sortGroupsForDisplay(groups) {
  const pinned = groups.filter((g) => g.pinned);
  const unpinned = groups.filter((g) => !g.pinned);
  const sortFn = (a, b) => (a.sortOrder || 0) - (b.sortOrder || 0);
  return [...pinned.sort(sortFn), ...unpinned.sort(sortFn)];
}

/**
 * 获取所有分组中的常用网页（带分组信息）。
 */
export function getFavoriteItems(groups) {
  const items = [];
  for (const group of groups) {
    for (const child of group.children) {
      if (child.kind === 'item' && child.favorite) {
        items.push({ ...child, groupId: group.id, groupName: group.name });
      }
    }
  }
  return items;
}
```

```js
import { createItem, addChildToGroup, normalizeState } from './schema.js';

export async function saveItemToGroup(chromeApi, groupId, input) {
  const state = normalizeState((await chromeApi.storage.local.get([STORAGE_KEY]))[STORAGE_KEY]);
  const item = createItem(input);

  // 检查同分组内是否有重复 URL
  const group = state.groups.find((g) => g.id === groupId);
  if (group && group.children.some((c) => c.kind === 'item' && c.url === item.url)) {
    return { item: null, skippedDuplicates: [item.url] };
  }

  const updated = addChildToGroup(state, groupId, { kind: 'item', ...item });
  delete updated.ui; // setState 内部会带 ui
  await setState(chromeApi, updated);
  return { item, skippedDuplicates: [] };
}
```

```js
 * 将子分组移动到另一个分组下（改 parentId + 在目标 children 中插入引用）。
 *
 * @param {Object} chromeApi - Chrome API
 * @param {string} groupId - 待移动的分组 ID
 * @param {string} newParentId - 新父分组 ID（null 表示移到根级）
 * @param {number} toIndex - 目标父分组 children 中的插入位置
 * @returns {Promise<Object>} 最新状态
 */
export async function moveGroupToParent(chromeApi, groupId, newParentId, toIndex) {
  const state = normalizeState((await chromeApi.storage.local.get([STORAGE_KEY]))[STORAGE_KEY]);
  const group = findGroup(state.groups, groupId);
  if (!group) return state;

  let s = state;

  // 从旧父分组的 children 中移除引用
  if (group.parentId) {
    const oldParent = findGroup(s.groups, group.parentId);
    if (oldParent) {
      s = {
        ...s,
        groups: s.groups.map((g) => {
          if (g.id !== group.parentId) return g;
          return {
            ...g,
            children: g.children.filter((c) => !(c.kind === 'group' && c.groupId === groupId)),
            updatedAt: generateTimestamp()
          };
        })
      };
    }
  }

  // 改 parentId
  s = updateGroup(s, groupId, { parentId: newParentId });

  // 在新父分组的 children 中插入引用
  if (newParentId) {
    s = {
      ...s,
      groups: s.groups.map((g) => {
        if (g.id !== newParentId) return g;
        return { ...g, children: [...g.children, { kind: 'group', groupId }], updatedAt: generateTimestamp() };
      })
    };
  }

  await setState(chromeApi, s);
  return s;
}
```

```js
import { findGroup, sortGroupsForDisplay } from '../shared/schema.js';

/**
 * 构建树形视图分组列表。仅返回根级分组，子分组通过其 children 中的 group 引用在渲染时递归展开。
 *
 * @param {Object} state - 完整状态
 * @param {string} keyword - 搜索关键字
 * @returns {Object[]} 根级分组数组
 */
export function buildTreeViewGroups(state, keyword = '') {
  // 仅返回根级分组
  const rootGroups = state.groups.filter((g) => g.parentId === null);
  const sorted = sortGroupsForDisplay(rootGroups);
  if (!keyword) return sorted;

  // 搜索时需要递归检查整个树
  const kw = keyword.toLowerCase();
  // 找出所有匹配的分组及其祖先
  const matchedIds = new Set();
  for (const group of state.groups) {
    const nameMatch = group.name.toLowerCase().includes(kw);
    const childMatch = group.children.some((c) => {
      if (c.kind === 'item') {
        return (c.title || '').toLowerCase().includes(kw) ||
               (c.url || '').toLowerCase().includes(kw) ||
               (getDisplayDomain(c.url) || '').toLowerCase().includes(kw);
      }
      return false;
    });
    if (nameMatch || childMatch) {
      matchedIds.add(group.id);
      // 标记所有祖先
      let current = group;
      while (current.parentId) {
        matchedIds.add(current.parentId);
        current = findGroup(state.groups, current.parentId);
        if (!current) break;
      }
    }
  }
  return sorted.filter((g) => matchedIds.has(g.id));
}

/**
 * 渲染时递归获取分组的所有子内容（children 按条目排列）。
 * 返回结构：{ group, children: [{...child, childGroup?}] }
 */
export function flattenGroupTree(groups, groupId, depth = 0) {
  const group = findGroup(groups, groupId);
  if (!group) return [];
  const result = [];
  for (const child of group.children) {
    if (child.kind === 'item') {
      result.push({ ...child, depth, parentGroupId: groupId });
    } else if (child.kind === 'group') {
      const childGroup = findGroup(groups, child.groupId);
      if (childGroup) {
        result.push({ kind: 'group-header', group: childGroup, depth });
        result.push(...flattenGroupTree(groups, child.groupId, depth + 1));
      }
    }
  }
  return result;
}
```

```js
function attachGroupDrag(element, group) {
  element.addEventListener('dragstart', (event) => {
    event.dataTransfer.setData('application/group-id', group.id);
    element.classList.add('dragging');
  });
  element.addEventListener('dragend', () => {
    element.classList.remove('dragging');
  });

  // 作为 drop 目标：接收其他分组或网页
  element.addEventListener('dragover', (event) => {
    event.preventDefault();
    event.dataTransfer.dropEffect = 'move';
    element.classList.add('drag-over');
  });
  element.addEventListener('dragleave', () => {
    element.classList.remove('drag-over');
  });
  element.addEventListener('drop', async (event) => {
    event.preventDefault();
    element.classList.remove('drag-over');
    const groupId = event.dataTransfer.getData('application/group-id');
    const itemData = event.dataTransfer.getData('application/item-data');
    if (groupId) {
      // 拖拽的是分组：移到当前分组下成为子分组
      if (groupId === group.id) return; // 不能放自己身上
      if (isDescendant(currentState.groups, groupId, group.id)) {
        showToast('不能将分组放入自己的子分组中');
        return;
      }
      currentState = await moveGroupToParent(chrome, groupId, group.id, group.children.length);
    } else if (itemData) {
      // 拖拽的是网页
      const { fromGroupId, fromIndex } = JSON.parse(itemData);
      currentState = await moveItemToGroupStorage(chrome, fromGroupId, fromIndex, group.id, group.children.length);
    }
    render();
  });
}

      fromIndex: childIndex
    }));
    element.classList.add('dragging');
  });
```
