~~~python
import contextlib
import hashlib
import subprocess
from pathlib import Path

def command(root, *args):
    return subprocess.check_output(["git", "-C", str(root), *args], text=True).strip()

def sha(value: bytes) -> str:
    return hashlib.sha256(value).hexdigest()

@contextlib.contextmanager
def login_overlay(root: Path, backup_dir: Path):
    root = root.resolve(strict=True)
    if Path(command(root, "rev-parse", "--show-toplevel")).resolve() != root:
        raise ValueError("登录覆盖必须定位真实 Git 根")
    target = root / "src/views/login.vue"
    source = root / "src/views/login公众号验证码登录.vue"
    for file in (target, source):
        if not file.resolve().is_relative_to(root) or file.is_symlink():
            raise ValueError("登录覆盖不能跨仓或通过符号链接")
        relative = str(file.relative_to(root))
        command(root, "ls-files", "--error-unmatch", relative)
        if command(root, "status", "--porcelain", "--", relative):
            raise ValueError("登录源或目标存在未提交内容")
    backup_dir = backup_dir.resolve()
    if backup_dir.is_relative_to(root):
        raise ValueError("备份必须在项目外")
    backup_dir.mkdir(parents=True, exist_ok=False)
    original, replacement = target.read_bytes(), source.read_bytes()
    backup = backup_dir / "login.original.bytes"
    backup.write_bytes(original)
    result = {"candidate_sha": command(root, "rev-parse", "HEAD"),
              "original_sha256": sha(original),
              "replacement_sha256": sha(replacement), "restored": False}
    try:
        target.write_bytes(replacement)
        if sha(target.read_bytes()) != result["replacement_sha256"]:
            raise RuntimeError("登录覆盖校验失败")
        yield result
    finally:
        target.write_bytes(backup.read_bytes())
        clean = not command(root, "status", "--porcelain", "--",
                            "src/views/login.vue")
        result["restored"] = sha(target.read_bytes()) == result["original_sha256"] and clean
        if not result["restored"]:
            raise RuntimeError("登录页恢复失败，保留外部备份并阻止放行")
~~~

以上写文件属于实施后受控运行操作；仓库代码本身仍用 apply_patch 新增。外层入口将 SIGTERM 转成受控中断后走 finally；SIGKILL、主机断电不能承诺即时恢复，外部备份用于恢复检查，未恢复不得报告通过。验收期间目标工作区不得交给另一个任务修改。

~~~python
from pathlib import Path, PurePosixPath

def assert_resources(evidence: dict) -> None:
    for key in ("dependencies_ready", "existing_consumer_boundary_verified",
                "our_consumer_boundary_verified", "file_mapping_verified"):
        if evidence.get(key) is not True:
            raise ValueError("真实计算安全前提缺失：" + key)

def r_to_java(r_file: str, mounts: list[dict], approved_roots: list[Path]) -> Path:
    r_path = PurePosixPath(r_file)
    if ".." in r_path.parts or not r_path.is_absolute():
        raise ValueError("非法 R 数据文件路径")
    matches = []
    for mount in mounts:
        target = PurePosixPath(mount["Destination"])
        if r_path.is_relative_to(target):
            matches.append((len(target.parts), mount, r_path.relative_to(target)))
    if not matches:
        raise ValueError("R 数据文件没有已核验挂载")
    _, mount, relative = max(matches, key=lambda item: item[0])
    if not mount.get("RW"):
        raise ValueError("结果目录不是可写业务挂载")
    path = (Path(mount["Source"]) / str(relative)).resolve()
    if not any(path.is_relative_to(root.resolve()) for root in approved_roots):
        raise ValueError("文件不在批准业务数据根，禁止读取")
    return path
~~~

~~~javascript
const assert = require('node:assert/strict')

function createWitness(page, systemOrigin, ownedSession) {
  const expected = new URL(systemOrigin)
  assert.ok(ownedSession && ownedSession.session_id && ownedSession.pid > 0)
  const frames = []
  let currentId
  let started = 0
  page.on('websocket', socket => {
    const origin = new URL(socket.url())
    if (origin.hostname !== '127.0.0.1' || origin.port !== expected.port) return
    socket.on('framereceived', frame => {
      let value
      try { value = JSON.parse(String(frame.payload)) } catch { return }
      if (value.noticeType !== 3 || value.status !== 4) return
      frames.push({ receivedAt: Date.now(), id: String(value.id),
        noticeType: value.noticeType, status: value.status,
        hasHtml: Boolean(value.data && value.data.userDraw &&
                         value.data.userDraw.html) })
    })
  })
  return {
    begin(drawId) {
      currentId = String(drawId)
      assert.match(currentId, /^\d+$/)
      started = Date.now()
      frames.length = 0
    },
    assertComplete(taskLog, taskPid) {
      assert.ok(Number.isInteger(taskPid) && taskPid > 0)
      assert.equal(taskPid, ownedSession.pid, '不是本次受控 task 会话')
      const success = frames.find(frame =>
        frame.id === currentId && frame.receivedAt >= started)
      assert.ok(success, '缺少本次绘图实际收到的成功 WebSocket 帧')
      const pattern = new RegExp(
        '开始处理消息: userDrawId=' + currentId +
        ', userId=\\d+, messageId=([0-9]+-[0-9]+)')
      const processed = String(taskLog).match(pattern)
      assert.ok(processed, '本次受控 task 日志没有处理该绘图 ID')
      assert.ok(new RegExp('处理的任务id:' + currentId + '(?![0-9])')
                .test(String(taskLog)), '本次窗口内未实际进入绘图执行线程')
      return { drawId: currentId, taskPid, messageId: processed[1],
               websocketReceived: true, status: success.status }
    }
  }
}

module.exports = { createWitness }
~~~
