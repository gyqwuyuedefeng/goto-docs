# 侧边栏导航器 UI 优化 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 将现有 Chrome 标签侧边栏管理器从顶部按钮列表面板升级为支持 G 徽章、favicon、树形列表、图标墙、折叠、拖拽宽度和主题的左侧导航器体验。

**Architecture:** 保持 Manifest V3 原生 JavaScript 实现，不引入构建工具。先扩展纯数据模型和存储层以支持 UI 偏好、置顶、常用和 favicon，再将 sidepanel 拆成可测试的 UI 状态/视图辅助模块，最后改造 HTML/CSS/JS 结构并通过浏览器手工验收。

**Tech Stack:** Chrome Extension Manifest V3、Chrome Side Panel API、`chrome.storage.local`、原生 HTML/CSS/JavaScript、Node.js `node:test`、现有 mock Chrome API。

```js
export const UI_DEFAULTS = {
  mode: 'tree',
  collapsed: false,
  navWidth: 340,
  theme: 'light'
};

export const NAV_WIDTH = {
  min: 240,
  max: 520,
  collapsed: 72
};

const VALID_UI_MODES = new Set(['tree', 'iconWall']);
const VALID_THEMES = new Set(['light', 'dark', 'gGradient']);

export function createDefaultUi(overrides = {}) {
  return {
    mode: VALID_UI_MODES.has(overrides.mode) ? overrides.mode : UI_DEFAULTS.mode,
    collapsed: typeof overrides.collapsed === 'boolean' ? overrides.collapsed : UI_DEFAULTS.collapsed,
    navWidth: clampNavWidth(overrides.navWidth),
    theme: VALID_THEMES.has(overrides.theme) ? overrides.theme : UI_DEFAULTS.theme
  };
}

export function clampNavWidth(value) {
  const width = Number.isFinite(value) ? value : UI_DEFAULTS.navWidth;
  return Math.min(NAV_WIDTH.max, Math.max(NAV_WIDTH.min, width));
}

export function getDomainFromUrl(url) {
  try {
    return new URL(url).hostname;
  } catch {
    return '';
  }
}

export function getFaviconUrl(url) {
  const domain = getDomainFromUrl(url);
  if (!domain) return '';
  return `https://www.google.com/s2/favicons?domain=${domain}&sz=32`;
}
```

```js
export function normalizeState(state) {
  const raw = state && typeof state === 'object' ? state : {};
  const groups = Array.isArray(raw.groups) ? raw.groups : [];
  return {
    ui: createDefaultUi(raw.ui || {}),
    groups: groups.filter((group) => group && typeof group.id === 'string').map((group) => ({
      id: group.id,
      name: typeof group.name === 'string' ? group.name : '未命名分组',
      pinned: typeof group.pinned === 'boolean' ? group.pinned : false,
      sortOrder: Number.isFinite(group.sortOrder) ? group.sortOrder : 0,
      expanded: typeof group.expanded === 'boolean' ? group.expanded : true,
      createdAt: group.createdAt || generateTimestamp(),
      updatedAt: group.updatedAt || group.createdAt || generateTimestamp(),
      items: Array.isArray(group.items)
        ? group.items
            .filter((item) => item && typeof item.id === 'string')
            .map((item) => ({
              id: item.id,
              title: typeof item.title === 'string' ? item.title : item.url || '未命名网页',
              url: typeof item.url === 'string' ? item.url : '',
              source: typeof item.source === 'string' ? item.source : 'page',
              favorite: typeof item.favorite === 'boolean' ? item.favorite : false,
              faviconUrl: typeof item.faviconUrl === 'string' && item.faviconUrl ? item.faviconUrl : getFaviconUrl(item.url),
              sortOrder: Number.isFinite(item.sortOrder) ? item.sortOrder : 0,
              createdAt: item.createdAt || generateTimestamp()
            }))
        : []
    }))
  };
}
```

```js
export function sortGroupsForDisplay(groups) {
  return [...groups].sort((a, b) => {
    if (a.pinned !== b.pinned) return a.pinned ? -1 : 1;
    return a.sortOrder - b.sortOrder;
  });
}

export function getFavoriteItems(groups) {
  return groups.flatMap((group) =>
    group.items
      .filter((item) => item.favorite)
      .map((item) => ({
        groupId: group.id,
        groupName: group.name,
        item
      }))
  );
}

export function updateUiPreferences(state, patches) {
  const normalized = normalizeState(state);
  return normalizeState({
    ...normalized,
    ui: createDefaultUi({
      ...normalized.ui,
      ...patches
    })
  });
}

export function toggleGroupPinned(state, groupId) {
  const normalized = normalizeState(state);
  return normalizeState({
    ...normalized,
    groups: normalized.groups.map((group) => {
      if (group.id !== groupId) return group;
      return { ...group, pinned: !group.pinned, updatedAt: generateTimestamp() };
    })
  });
}

export function toggleItemFavorite(state, groupId, itemId) {
  const normalized = normalizeState(state);
  return normalizeState({
    ...normalized,
    groups: normalized.groups.map((group) => {
      if (group.id !== groupId) return group;
      return {
        ...group,
        updatedAt: generateTimestamp(),
        items: group.items.map((item) => {
          if (item.id !== itemId) return item;
          return { ...item, favorite: !item.favorite };
        })
      };
    })
  });
}
```

```js
import { getDomainFromUrl, getFavoriteItems, sortGroupsForDisplay } from '../shared/schema.js';

/**
 * 侧边栏视图模型辅助函数。
 *
 * 本模块只做展示数据计算，不访问 DOM 和 Chrome API。这样树形列表、
 * 图标墙、折叠快捷列和主题 class 的规则都可以在 Node.js 中测试。
 */

export function getDisplayDomain(url) {
  return getDomainFromUrl(url);
}

export function buildTreeViewGroups(state, keyword = '') {
  const normalizedKeyword = keyword.trim().toLowerCase();
  return sortGroupsForDisplay(state.groups)
    .map((group) => {
      const items = group.items
        .map((item) => ({
          ...item,
          domain: getDisplayDomain(item.url)
        }))
        .filter((item) => matchesKeyword(group, item, normalizedKeyword));
      return { ...group, items };
    })
    .filter((group) => {
      if (!normalizedKeyword) return true;
      return group.name.toLowerCase().includes(normalizedKeyword) || group.items.length > 0;
    });
}

export function buildIconWallSections(state, keyword = '') {
  const normalizedKeyword = keyword.trim().toLowerCase();
  const favoriteItems = getFavoriteItems(state.groups)
    .map(toIconEntry)
    .filter((entry) => matchesEntryKeyword(entry, normalizedKeyword));

  const sections = [];
  if (favoriteItems.length > 0) {
    sections.push({
      id: 'favorites',
      title: '常用',
      items: favoriteItems
    });
  }

  for (const group of sortGroupsForDisplay(state.groups)) {
    const items = group.items
      .map((item) => toIconEntry({ groupId: group.id, groupName: group.name, item }))
      .filter((entry) => matchesEntryKeyword(entry, normalizedKeyword));
    if (!normalizedKeyword || items.length > 0 || group.name.toLowerCase().includes(normalizedKeyword)) {
      sections.push({
        id: `group-${group.id}`,
        groupId: group.id,
        title: group.name,
        items
      });
    }
  }

  return sections;
}

export function buildCollapsedShortcutItems(state, limit = 12) {
  const favorites = getFavoriteItems(state.groups).map(toIconEntry);
  if (favorites.length > 0) return favorites.slice(0, limit);

  return sortGroupsForDisplay(state.groups)
    .flatMap((group) =>
      group.items.map((item) => toIconEntry({ groupId: group.id, groupName: group.name, item }))
    )
    .slice(0, limit);
}

export function getThemeClass(theme) {
  if (theme === 'dark') return 'theme-dark';
  if (theme === 'gGradient') return 'theme-g-gradient';
  return 'theme-light';
}

function toIconEntry(entry) {
  return {
    ...entry,
    domain: getDisplayDomain(entry.item.url)
  };
}

function matchesKeyword(group, item, keyword) {
  if (!keyword) return true;
  return group.name.toLowerCase().includes(keyword) ||
    item.title.toLowerCase().includes(keyword) ||
    item.url.toLowerCase().includes(keyword) ||
    item.domain.toLowerCase().includes(keyword);
}

function matchesEntryKeyword(entry, keyword) {
  if (!keyword) return true;
  return entry.groupName.toLowerCase().includes(keyword) ||
    entry.item.title.toLowerCase().includes(keyword) ||
    entry.item.url.toLowerCase().includes(keyword) ||
    entry.domain.toLowerCase().includes(keyword);
}
```
