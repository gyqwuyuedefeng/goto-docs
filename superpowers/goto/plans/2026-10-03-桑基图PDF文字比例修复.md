```java
/**
 * 为 HTML 交互图构造既有 SVG 转换命令。
 * 桑基图 SVG 的文字使用 pt，节点坐标使用 viewBox 单位；PDF 若沿用 300 DPI，
 * 页面坐标会缩小而文字保持物理字号，故仅此格式按浏览器的 96 DPI 换算。
 * 网络图及其他格式不传 -d，继续遵守转换脚本原有默认值。
 */
String[] buildFileConvertCommand(String inputPath, String outputPath, String imageType) {
    List<String> command = new ArrayList<>(List.of(
            "sh",
            componentBean.getProperties().getConfigProperties().getR().getFileConvertScript(),
            "-i", inputPath,
            "-o", outputPath
    ));
    if (plotType.getGroupId() == 15L && "pdf".equalsIgnoreCase(imageType)) {
        command.add("-d");
        command.add("96");
    }
    return command.toArray(new String[0]);
}
```
